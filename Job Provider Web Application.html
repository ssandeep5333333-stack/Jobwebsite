Node.js + Express + SQLite Backend Reference

// server.js - Express backend API

const express = require('express');
const sqlite3 = require('sqlite3').verbose();
const cors = require('cors');

const app = express();
app.use(express.json());
app.use(cors());

const db = new sqlite3.Database('./job_provider.db');

// Table Initialization
db.serialize(() => {
  db.run(`CREATE TABLE IF NOT EXISTS registrations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    registration_id TEXT UNIQUE,
    full_name TEXT,
    mobile TEXT,
    email TEXT,
    city TEXT,
    state TEXT,
    age INTEGER,
    preferred_category TEXT,
    experience TEXT,
    whatsapp TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS payments (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    registration_id TEXT,
    amount INTEGER DEFAULT 100,
    transaction_id TEXT,
    status TEXT DEFAULT 'Pending Verification',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    verified_at DATETIME
  )`);
});

// API Routes
app.post('/api/register', (req, res) => {
  const { full_name, mobile, email, city, state, age, preferred_category, experience, whatsapp } = req.body;
  const regId = 'JP-' + new Date().getFullYear() + '-' + Math.floor(1000 + Math.random() * 9000);
  
  db.run(`INSERT INTO registrations (registration_id, full_name, mobile, email, city, state, age, preferred_category, experience, whatsapp)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
    [regId, full_name, mobile, email, city, state, age, preferred_category, experience, whatsapp],
    function(err) {
      if (err) return res.status(500).json({ error: err.message });
      res.json({ success: true, registration_id: regId });
    });
});

app.post('/api/payment-submit', (req, res) => {
  const { registration_id, transaction_id } = req.body;
  db.run(`INSERT INTO payments (registration_id, transaction_id, status) VALUES (?, ?, 'Pending Verification')`,
    [registration_id, transaction_id],
    function(err) {
      if (err) return res.status(500).json({ error: err.message });
      res.json({ success: true, message: 'Payment submitted for verification' });
    });
});

app.listen(3000, () => console.log('Job Provider Backend running on port 3000'));
